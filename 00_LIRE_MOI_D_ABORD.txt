💡 EXEMPLE D'UTILISATION
═══════════════════════════════════════════════════════════════════════════════

from stripe_integration import PaymentService, SUBSCRIPTION_PLANS

# Créer un client Stripe
customer_id = PaymentService.create_customer(
    email='client@example.com',
    name='Jean Dupont'
)

# Créer un abonnement avec 30 jours gratuits
subscription = PaymentService.create_subscription(
    customer_id=customer_id,
    plan='starter',  # starter, professional, ou enterprise
    trial_days=30
)

print(f"Abonnement créé: {subscription['subscription_id']}")
print(f"Essai gratuit jusqu'au: {subscription['trial_end']}")
